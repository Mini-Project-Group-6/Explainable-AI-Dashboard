# Dashboard (S3) dependencies.
#
# Direct dependencies only. This file was previously a `pip freeze` of a
# different interpreter, which pinned 48 transitive packages including
# numpy==2.4.6 and pandas==3.0.3. Installing that into the model venv upgraded
# numpy past 2.0 and broke xgboost 1.7.6 — model/requirements.txt holds numpy
# below 2.0 deliberately, because xgboost 1.7.6 predates the numpy 2 ABI.
#
# So numpy, pandas, scipy and click are NOT pinned here (scipy, which the S5
# analysis uses, arrives with scikit-learn). They are shared with the
# model layer and model/requirements.txt is where their versions are decided.
# Streamlit and Altair work fine against those older pins; the two stacks were
# installed side by side and verified.
#
# Setup — one venv for both layers, model stack first so its pins win:
#   py -3.12 -m venv .venv
#   .venv\Scripts\python -m pip install -r model/requirements.txt
#   .venv\Scripts\python -m pip install torch --index-url https://download.pytorch.org/whl/cpu
#   .venv\Scripts\python -m spacy download en_core_web_sm
#   .venv\Scripts\python -m pip install -r requirements.txt
#
# Run:
#   .venv\Scripts\python -m streamlit run main.py

streamlit==1.58.0
altair==6.2.2

# Storage. psycopg2-binary rather than psycopg2 so no local libpq/compiler is
# needed on the deployment laptops.
SQLAlchemy==2.0.51
psycopg2-binary==2.9.12
python-dotenv==1.2.2
